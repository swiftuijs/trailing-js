 const ratios=pairs.map(p=>p.rust[key]/p.native[key]);
 let seed=123456789;
 const random=()=> (seed=(Math.imul(seed,1664525)+1013904223)>>>0)/4294967296;
 const resamples=Array.from({length:10000},()=>median(Array.from({length:ratios.length},()=>ratios[Math.floor(random()*ratios.length)]))).sort((a,b)=>a-b);
 return {medianRatio:median(ratios),ratio95PercentInterval:{method:'paired bootstrap median, 10000 deterministic resamples',lower:resamples[250],upper:resamples[9750]}};

async function operation(run,w) {
 const usage=process.cpuUsage(),start=performance.now();
 const result=await run(w.command,w.options);
 const wallMs=performance.now()-start,cpu=process.cpuUsage(usage);
 assert.deepEqual(result.terminationStatus,{kind:'exited',code:0});
 if(w.expected) assert(result.standardOutput.equals(w.expected));
 if(w.expectedError) assert(result.standardError.equals(w.expectedError));
 return {wallMs,parentCPUMs:(cpu.user+cpu.system)/1000};
}

 const pairs=[];
 for(let sample=0;sample<12;sample++) {
  const values=Object.fromEntries(Object.keys(modes).map(mode=>[mode,{wallMs:0,parentCPUMs:0}]));
  const operationOrders=[];
  for(let n=0;n<w.count;n++) {
   const order=orders[(sample+n)%orders.length]; operationOrders.push(order);
   for(const mode of order) {
    const result=await operation(modes[mode],w);
    values[mode].wallMs+=result.wallMs/w.count;
    values[mode].parentCPUMs+=result.parentCPUMs/w.count;
   }
  }
  pairs.push({sample,operationOrders,...values});
 }

}
