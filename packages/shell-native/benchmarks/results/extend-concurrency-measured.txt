import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {execFileSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import {resolve} from 'node:path';
import {fileURLToPath} from 'node:url';
const directory='/workspace/twill-shell-native/packages/shell-native/benchmarks';
const [input,output]=process.argv.slice(2);
assert(input&&output,'Expected complete input report and output path');
const source=readFileSync(input),report=JSON.parse(source);
const sha=bytes=>createHash('sha256').update(bytes).digest('hex');
for(const [name,digest] of Object.entries(report.sourceAndBuildSHA256))assert.equal(sha(readFileSync(resolve(directory,name))),digest,'Measured source/build changed: '+name);
assert.equal(process.version,report.node,'Node runtime changed');
assert.equal(readFileSync('/sys/fs/cgroup/cpu.max','utf8').trim(),report.cpuConstraints.cgroupV2Quota,'CPU quota changed');
const observer=resolve(directory,'observe.mjs');
const median=values=>[...values].sort((a,b)=>a-b)[Math.floor(values.length/2)];
function comparison(pairs,baseline){
 const ratios=pairs.map(pair=>pair.rust.wallMs/pair[baseline].wallMs);let seed=123456789;
 const random=()=>((seed=(Math.imul(seed,1664525)+1013904223)>>>0)/4294967296);
 const medians=Array.from({length:10000},()=>median(Array.from({length:ratios.length},()=>ratios[Math.floor(random()*ratios.length)]))).sort((a,b)=>a-b);
 return {medianRatio:median(ratios),ratio95PercentInterval:{method:'paired bootstrap median, 10000 deterministic resamples',lower:medians[250],upper:medians[9750]}};
}
report.concurrencyExtension={dateUTC:new Date().toISOString(),baseReportSHA256:sha(source),scriptSHA256:sha(readFileSync(fileURLToPath(import.meta.url))),gitHead:execFileSync('git',['rev-parse','HEAD'],{cwd:resolve(directory,'../../..'),encoding:'utf8'}).trim(),samplesPerDefaultPoolCase:48,sourceAndBuildRevalidated:true,scope:'Extend both default-pool concurrency cases to 48 balanced paired samples on the exact previously measured build. Preserve every original pair and all previously completed warm/memory/cold/cancellation/filesystem observations; no discarded outliers, changed binaries, looser gates or repeated successful warm workloads.'};
for(const row of report.concurrency.filter(row=>row.poolSize==='4')){
 assert.equal(row.pairs.length,12,'Expected complete initial 12 pairs');
 for(let sample=row.pairs.length;sample<48;sample++){
  const pair={extensionSample:sample,order:report.orderPermutations[sample%6]};
  for(const mode of pair.order){
   pair[mode]=JSON.parse(execFileSync('taskset',['-c',report.concurrencyCPUs,process.execPath,observer,mode,'concurrency',String(row.children)],{encoding:'utf8',env:{...process.env,UV_THREADPOOL_SIZE:row.poolSize}}));
   assert.equal(pair[mode].allowedCPUList,report.concurrencyCPUs,'CPU affinity changed');
  }
  row.pairs.push(pair);row.rustVersusSDK={wall:comparison(row.pairs,'sdk')};row.rustVersusNative={wall:comparison(row.pairs,'native')};
  writeFileSync(output,JSON.stringify(report,null,2)+'\n');
 }
 console.log(row.children,row.rustVersusNative.wall);
}
report.concurrencyExtension.completedUTC=new Date().toISOString();
writeFileSync(output,JSON.stringify(report,null,2)+'\n');
for(const row of [...report.results,...report.concurrency.filter(row=>row.poolSize==='4')])assert(row.rustVersusNative.wall.medianRatio<=1.1&&row.rustVersusNative.wall.ratio95PercentInterval.upper<=1.1,'Parity gate failed: '+(row.name??row.children));
console.log('All unchanged 1.10 median and upper-confidence gates passed');
